# Source execution policy depends on the shared private CLI support inventory.
set(_run_root "${CMAKE_BINARY_DIR}/lib/beam/runtime/beam/run")
set(_run_out "${_run_root}/ebin")
set(_run_include "${CMAKE_BINARY_DIR}/run/include")
set(BEAM_RUN_MODULES beam_source_run)
set(BEAM_RUN_BEAM_FILES "${_run_out}/beam_source_run.beam")
beam_add_erlang_component(beam_run_support
    SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/beam_source_run.erl"
    OUTPUT_DIRECTORY "${_run_out}")
add_dependencies(beam_run_support beam_cli_support)
file(MAKE_DIRECTORY "${_run_include}")
set(_inventory "/* Private matching-build source-run inventory. */\nstatic const char *const beam_run_modules[] = {\n")
foreach(_module IN LISTS BEAM_RUN_MODULES)
    string(APPEND _inventory "    \"${_module}\",\n")
endforeach()
list(LENGTH BEAM_RUN_MODULES _count)
string(APPEND _inventory "};\n#define BEAM_RUN_MODULE_COUNT ${_count}\n")
file(CONFIGURE OUTPUT "${_run_include}/beam_run_inventory.h" CONTENT "${_inventory}" NEWLINE_STYLE UNIX @ONLY)
string(JOIN "\n" _manifest ${BEAM_RUN_MODULES})
file(CONFIGURE OUTPUT "${_run_root}/modules.txt" CONTENT "${_manifest}\n" NEWLINE_STYLE UNIX @ONLY)
install(FILES ${BEAM_RUN_BEAM_FILES} DESTINATION lib/beam/runtime/beam/run/ebin)
install(FILES "${_run_root}/modules.txt" DESTINATION lib/beam/runtime/beam/run)
set(BEAM_RUN_INCLUDE_DIRECTORY "${_run_include}" PARENT_SCOPE)
foreach(_value IN ITEMS MODULES BEAM_FILES)
    set(BEAM_RUN_${_value} "${BEAM_RUN_${_value}}" PARENT_SCOPE)
endforeach()
