cmake_minimum_required(VERSION 3.22)
project(BEAMAndroidAcceptance LANGUAGES C CXX)
if(NOT BEAM_INPUTS OR NOT ANDROID)
    message(FATAL_ERROR "Select installed Android application inputs")
endif()
set(BEAM_DIR "${BEAM_INPUTS}/sdk/${ANDROID_ABI}/lib/cmake/BEAM")
set(BEAM_ANDROID_PREFAB ON)
find_package(BEAM CONFIG REQUIRED COMPONENTS Runtime NIF)
include("${BEAM_INPUTS}/exports/${ANDROID_ABI}/BEAMPackageExport.cmake")
add_library(beam_acceptance SHARED owner.cpp)
target_compile_features(beam_acceptance PRIVATE cxx_std_17)
target_link_libraries(beam_acceptance PRIVATE BEAMPackageInputs)
set(_mask 0)
foreach(_component IN LISTS BEAM_FIXTURE_COMPONENTS)
    if(_component STREQUAL "file")
        math(EXPR _mask "${_mask} | 1")
    elseif(_component STREQUAL "network")
        math(EXPR _mask "${_mask} | 2")
    elseif(_component STREQUAL "zlib")
        math(EXPR _mask "${_mask} | 4")
    else()
        message(FATAL_ERROR "Unsupported fixture component ${_component}")
    endif()
endforeach()
target_compile_definitions(beam_acceptance PRIVATE BEAM_FIXTURE_COMPONENTS=${_mask})
set(_components)
if(BEAM_FIXTURE_COMPONENTS)
    set(_components COMPONENTS ${BEAM_FIXTURE_COMPONENTS})
endif()
beam_target_android_application(beam_acceptance ${_components} INVENTORIES base services
    NATIVE_TARGETS ${BEAM_PACKAGE_NATIVE_TARGETS})

foreach(_asset IN LISTS BEAM_PACKAGE_ASSETS)
    if(_asset MATCHES "/priv/payload.txt$")
        target_compile_definitions(beam_acceptance PRIVATE "FIXTURE_RESOURCE=\"${_asset}\"")
    endif()
endforeach()
