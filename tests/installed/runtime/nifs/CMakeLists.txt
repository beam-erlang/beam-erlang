cmake_minimum_required(VERSION 3.25)
project(native_acceptance LANGUAGES C)
set(RUNTIME_COMPONENT Runtime CACHE STRING "Selected SDK runtime linkage")
find_package(BEAM CONFIG REQUIRED COMPONENTS ${RUNTIME_COMPONENT})
include("${TEST_SUPPORT_DIR}/HostObservation.cmake")
set(_origin "$ORIGIN/../lib")
if(APPLE)
    set(_origin "@executable_path/../lib")
endif()
foreach(name probe fail_load bad_abi bad_module bad_function unstamped wrongstamp)
    add_library(${name} MODULE probe.c)
    if(name STREQUAL "unstamped" OR name STREQUAL "wrongstamp")
        target_link_libraries(${name} PRIVATE BEAM::NIF)
    else()
        beam_target_nif(${name})
    endif()
    if(NOT name STREQUAL "probe" AND NOT name STREQUAL "unstamped" AND NOT name STREQUAL "wrongstamp")
        string(TOUPPER "${name}" flag)
        target_compile_definitions(${name} PRIVATE ${flag})
    endif()
    set_target_properties(${name} PROPERTIES PREFIX "")
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/benign.c" "int benign_dependency(void) { return 1; }\n")
    add_library(benign_dependency SHARED "${CMAKE_CURRENT_BINARY_DIR}/benign.c")
    add_library(search_dependency SHARED "${CMAKE_CURRENT_BINARY_DIR}/benign.c")
    foreach(name probe_sysv probe_gnu probe_both probe_dependency probe_search probe_cycle)
        add_library(${name} MODULE probe.c)
        beam_target_nif(${name})
        set_target_properties(${name} PROPERTIES PREFIX "")
    endforeach()
    target_link_options(probe_sysv PRIVATE "LINKER:--hash-style=sysv")
    target_link_options(probe_gnu PRIVATE "LINKER:--hash-style=gnu")
    target_link_options(probe_both PRIVATE "LINKER:--hash-style=both")
    target_link_options(probe_dependency PRIVATE "LINKER:--no-as-needed")
    target_link_libraries(probe_dependency PRIVATE benign_dependency)
    set_target_properties(probe_dependency PROPERTIES BUILD_RPATH "$ORIGIN")
    target_link_options(probe_search PRIVATE "LINKER:--no-as-needed")
    target_link_libraries(probe_search PRIVATE search_dependency)
    set_target_properties(probe_search PROPERTIES SKIP_BUILD_RPATH TRUE)
    # Link a real dependency cycle in two passes. The seed has the same SONAME
    # as the final library, but must never be selected by the runtime search.
    add_library(cycle_a_seed SHARED "${CMAKE_CURRENT_BINARY_DIR}/benign.c")
    set_target_properties(cycle_a_seed PROPERTIES OUTPUT_NAME cycle_a LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/seed")
    add_library(cycle_b SHARED "${CMAKE_CURRENT_BINARY_DIR}/benign.c")
    add_library(cycle_a SHARED "${CMAKE_CURRENT_BINARY_DIR}/benign.c")
    foreach(name cycle_a cycle_b probe_cycle)
        target_link_options(${name} PRIVATE "LINKER:--no-as-needed")
        set_target_properties(${name} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    endforeach()
    target_link_libraries(cycle_b PRIVATE cycle_a_seed)
    target_link_libraries(cycle_a PRIVATE cycle_b)
    target_link_libraries(probe_cycle PRIVATE cycle_a)
    if(RUNTIME_COMPONENT STREQUAL "Runtime")
        add_library(probe_runtime MODULE probe.c)
        beam_target_nif(probe_runtime)
        target_link_options(probe_runtime PRIVATE "LINKER:--no-as-needed")
        target_link_libraries(probe_runtime PRIVATE BEAM::Runtime)
        set_target_properties(probe_runtime PROPERTIES PREFIX "")
    endif()
endif()
add_executable(native_acceptance main.c)
target_link_libraries(native_acceptance PRIVATE BEAM::${RUNTIME_COMPONENT})
beam_test_observe(native_acceptance)
set_target_properties(native_acceptance PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${_origin}")

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/wrongstamp.c" "const char beam_nif_sdk_v1[] = \"different-sdk\";\n")
target_sources(wrongstamp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/wrongstamp.c")

if(RUNTIME_COMPONENT STREQUAL "RuntimeStatic")
    # Deliberately invalid extensions: a second copy of the selected runtime,
    # including through a transitive dependency. Initialization must not run.
    add_library(second_runtime MODULE probe.c)
    beam_target_nif(second_runtime)
    if(APPLE)
        target_link_options(second_runtime PRIVATE "LINKER:-needed_library,$<TARGET_FILE:BEAM::Runtime>")
    else()
        target_link_options(second_runtime PRIVATE "LINKER:--no-as-needed")
    endif()
    target_link_libraries(second_runtime PRIVATE BEAM::Runtime)
    set_target_properties(second_runtime PROPERTIES PREFIX "")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dependency.c" "int native_dependency(void) { return 1; }\n")
    add_library(runtime_dependency SHARED "${CMAKE_CURRENT_BINARY_DIR}/dependency.c")
    if(APPLE)
        target_link_options(runtime_dependency PRIVATE "LINKER:-needed_library,$<TARGET_FILE:BEAM::Runtime>")
    else()
        target_link_options(runtime_dependency PRIVATE "LINKER:--no-as-needed")
    endif()
    target_link_libraries(runtime_dependency PRIVATE BEAM::Runtime)
    add_library(transitive_runtime MODULE probe.c)
    beam_target_nif(transitive_runtime)
    if(APPLE)
        target_link_options(transitive_runtime PRIVATE "LINKER:-needed_library,$<TARGET_FILE:runtime_dependency>")
    else()
        target_link_options(transitive_runtime PRIVATE "LINKER:--no-as-needed")
    endif()
    target_link_libraries(transitive_runtime PRIVATE runtime_dependency)
    set_target_properties(transitive_runtime PROPERTIES PREFIX "")
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        # Runtime identity is a symbol property, independent of the filename.
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/foreign.c" "#include <stdio.h>\n#include <stdlib.h>\nconst char *beam_build_id(void) { return \"foreign\"; }\n__attribute__((constructor)) static void observed_constructor(void) { const char *p=getenv(\"NIF_CONSTRUCTOR_LOG\"); if(p) { FILE *f=fopen(p,\"a\"); if(f) { fputs(\"constructed\\n\",f); fclose(f); } } }\n")
        add_library(innocent SHARED "${CMAKE_CURRENT_BINARY_DIR}/foreign.c")
        foreach(style sysv gnu)
            add_library(identity_${style} MODULE probe.c "${CMAKE_CURRENT_BINARY_DIR}/foreign.c")
            beam_target_nif(identity_${style})
            target_link_options(identity_${style} PRIVATE "LINKER:--hash-style=${style}")
            set_target_properties(identity_${style} PROPERTIES PREFIX "")
        endforeach()
        add_library(renamed_runtime MODULE probe.c)
        beam_target_nif(renamed_runtime)
        target_link_options(renamed_runtime PRIVATE "LINKER:--no-as-needed")
        target_link_libraries(renamed_runtime PRIVATE innocent)
        set_target_properties(renamed_runtime PROPERTIES PREFIX "")
    endif()
endif()

target_link_libraries(builtin_acceptance PRIVATE BEAM::${RUNTIME_COMPONENT} BEAM::NIF BEAM::Driver)
set_target_properties(builtin_acceptance PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${_origin}")
