# Capture explicit Erlang dependencies at build time. The acceptance runner
# removes the installed runtime files before executing any bare host.
set(_support_header "${CMAKE_CURRENT_BINARY_DIR}/support.h")
file(WRITE "${_support_header}" "/* Embedded installed Erlang support modules. */\n")
function(embed_module symbol path)
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Required installed module is missing: ${path}")
    endif()
    file(READ "${path}" _bytes HEX)
    string(REGEX REPLACE "(..)" "0x\\1," _bytes "${_bytes}")
    file(APPEND "${_support_header}"
        "static const unsigned char ${symbol}[] = {${_bytes}};\n")
endfunction()
embed_module(host_module "${BEAM_HOST_MODULE}")
set(_io_modules io io_lib io_lib_format lists maps unicode)
foreach(_module IN LISTS _io_modules)
    embed_module(stdlib_${_module} "${CLI_EBIN_DIR}/${_module}.beam")
endforeach()
file(APPEND "${_support_header}" "static const beam_bytes io_modules[] = {\n")
foreach(_module IN LISTS _io_modules)
    file(APPEND "${_support_header}"
        "{stdlib_${_module}, sizeof stdlib_${_module}},\n")
endforeach()
file(APPEND "${_support_header}" "};\n")
