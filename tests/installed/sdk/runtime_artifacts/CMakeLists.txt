cmake_minimum_required(VERSION 3.25)
project(installed_runtime_artifact_consumer LANGUAGES C)
if(STATIC_PROBE)
    find_package(BEAM CONFIG REQUIRED COMPONENTS RuntimeStatic)
    find_package(ZLIB REQUIRED)
    find_library(PCRE2_LIBRARY NAMES pcre2-8 REQUIRED)
    find_path(PCRE2_INCLUDE_DIR pcre2.h REQUIRED)
    add_executable(static_probe static_probe.c)
    set(_pcre_runtime "${PCRE2_LIBRARY}")
    if(APPLE AND PCRE2_LIBRARY MATCHES "[.]tbd$")
        file(STRINGS "${PCRE2_LIBRARY}" _install_name REGEX "^install-name:")
        string(REGEX REPLACE "^install-name:[ ]*'([^']+)'.*$" "\\1" _pcre_runtime "${_install_name}")
        if(NOT _pcre_runtime MATCHES "^/.*[.]dylib$")
            message(FATAL_ERROR "Cannot resolve the selected PCRE SDK stub's runtime install name")
        endif()
    endif()
    target_compile_definitions(static_probe PRIVATE SYSTEM_PCRE_LIBRARY="${_pcre_runtime}")
    if(APPLE)
        target_sources(static_probe PRIVATE "${TEST_SUPPORT_DIR}/host_observe.c")
        target_include_directories(static_probe PRIVATE "${TEST_SUPPORT_DIR}")
    endif()
    target_include_directories(static_probe PRIVATE "${PCRE2_INCLUDE_DIR}")
    target_link_libraries(static_probe PRIVATE BEAM::RuntimeStatic BEAM::SDKSupport BEAM::Driver ZLIB::ZLIB "${PCRE2_LIBRARY}")
    beam_target_bytecode_verifier(static_probe INVENTORIES stdlib)
    if(NOT BEAM_STDLIB_MODULES STREQUAL "array;dict;graph;gb_sets;gb_trees;orddict;ordsets;proplists;queue;sets;sofs;digraph;digraph_utils"
       OR NOT EXISTS "${BEAM_STDLIB_MANIFEST}" OR NOT IS_DIRECTORY "${BEAM_STDLIB_EBIN_DIR}")
        message(FATAL_ERROR "Incomplete standard-library SDK discovery")
    endif()
    if(DRIVER_PROBE_BEAM)
        if(NOT BASE_EBIN_DIR)
            message(FATAL_ERROR "Driver execution requires selected installed base bytecode")
        endif()
        set(_driver_header "${CMAKE_CURRENT_BINARY_DIR}/driver_bytecode.h")
        file(WRITE "${_driver_header}" "/* Explicit installed bytecode for the process-owned driver check. */\n")
        set(_driver_modules beam beam_lists beam_unicode beam_runtime_unicode_util
            beam_math beam_ddll beam_debugger driver_probe)
        foreach(_module IN LISTS _driver_modules)
            if(_module STREQUAL driver_probe)
                set(_path "${DRIVER_PROBE_BEAM}")
            else()
                set(_path "${BASE_EBIN_DIR}/${_module}.beam")
            endif()
            file(READ "${_path}" _bytes HEX)
            string(REGEX REPLACE "(..)" "0x\\1," _bytes "${_bytes}")
            file(APPEND "${_driver_header}" "static const unsigned char driver_${_module}[] = {${_bytes}};\n")
        endforeach()
        file(APPEND "${_driver_header}" "static const beam_module_spec driver_modules[] = {\n")
        foreach(_module IN LISTS _driver_modules)
            file(APPEND "${_driver_header}"
                "{\"${_module}\", {driver_${_module}, sizeof driver_${_module}}, \"copied:installed-driver-check\"},\n")
        endforeach()
        file(APPEND "${_driver_header}" "};\n")
        target_include_directories(static_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${TEST_SUPPORT_DIR}")
        target_compile_definitions(static_probe PRIVATE BEAM_TEST_DRIVER_PROBE)
    endif()
    add_library(native_driver_probe MODULE driver_probe.c)
    target_link_libraries(native_driver_probe PRIVATE BEAM::Driver)
    set_target_properties(native_driver_probe PROPERTIES PREFIX "" OUTPUT_NAME sdk_driver_probe)
    if(MIXED_LINKAGE)
        target_link_libraries(static_probe PRIVATE BEAM::Runtime)
    endif()
    beam_install_core_modules(DESTINATION core/ebin)
    return()
endif()

    "${BEAM_SDK_BUILD_ID}\n${BEAM_SDK_FLAVOR}\n${selected}\n${relative}\n")
