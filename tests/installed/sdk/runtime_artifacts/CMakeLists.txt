cmake_minimum_required(VERSION 3.25)
project(installed_runtime_artifact_consumer LANGUAGES C)
if(STATIC_PROBE)
    find_package(BEAM CONFIG REQUIRED COMPONENTS RuntimeStatic)
    find_package(ZLIB REQUIRED)
    find_library(PCRE2_LIBRARY NAMES pcre2-8 REQUIRED)
    find_path(PCRE2_INCLUDE_DIR pcre2.h REQUIRED)
    add_executable(static_probe static_probe.c)
    set(_pcre_runtime "${PCRE2_LIBRARY}")
    if(APPLE AND PCRE2_LIBRARY MATCHES "[.]tbd$")
        file(STRINGS "${PCRE2_LIBRARY}" _install_name REGEX "^install-name:")
        string(REGEX REPLACE "^install-name:[ ]*'([^']+)'.*$" "\\1" _pcre_runtime "${_install_name}")
        if(NOT _pcre_runtime MATCHES "^/.*[.]dylib$")
            message(FATAL_ERROR "Cannot resolve the selected PCRE SDK stub's runtime install name")
        endif()
    endif()
    target_compile_definitions(static_probe PRIVATE SYSTEM_PCRE_LIBRARY="${_pcre_runtime}")
    if(APPLE)
        target_sources(static_probe PRIVATE "${TEST_SUPPORT_DIR}/host_observe.c")
        target_include_directories(static_probe PRIVATE "${TEST_SUPPORT_DIR}")
    endif()
    target_include_directories(static_probe PRIVATE "${PCRE2_INCLUDE_DIR}")
    target_link_libraries(static_probe PRIVATE BEAM::RuntimeStatic BEAM::Driver ZLIB::ZLIB "${PCRE2_LIBRARY}")
    add_library(native_driver_probe MODULE driver_probe.c)
    target_link_libraries(native_driver_probe PRIVATE BEAM::Driver)
    set_target_properties(native_driver_probe PROPERTIES PREFIX "" OUTPUT_NAME sdk_driver_probe)
    if(MIXED_LINKAGE)
        target_link_libraries(static_probe PRIVATE BEAM::Runtime)
    endif()
    beam_install_core_modules(DESTINATION core/ebin)
    return()
endif()
find_package(BEAM CONFIG REQUIRED)
beam_check_runtime_sdk(VERSION "${EXPECTED_VERSION}" FLAVOR "${EXPECTED_FLAVOR}")
if(DEFINED EXPECTED_BUILD_ID)
    beam_check_runtime_sdk(BUILD_ID "${EXPECTED_BUILD_ID}")
endif()
if(NOT DEFINED QUERY_ARTIFACT)
    set(QUERY_ARTIFACT runtime)
endif()
if(NOT DEFINED QUERY_KIND)
    set(QUERY_KIND runtime)
endif()
beam_get_runtime_artifact(selected NAME "${QUERY_ARTIFACT}" KIND "${QUERY_KIND}")
beam_get_runtime_artifact(relative NAME "${QUERY_ARTIFACT}" RELATIVE)
file(WRITE "${CMAKE_BINARY_DIR}/selection.txt"
    "${BEAM_SDK_BUILD_ID}\n${BEAM_SDK_FLAVOR}\n${selected}\n${relative}\n")
beam_install_runtime_artifacts(ARTIFACTS runtime helper.child_setup)

beam_install_native_headers(DESTINATION usr/include)
add_library(native_driver_probe MODULE driver_probe.c)
target_link_libraries(native_driver_probe PRIVATE BEAM::Driver)
set_target_properties(native_driver_probe PROPERTIES PREFIX "" OUTPUT_NAME sdk_driver_probe)

beam_install_core_modules(DESTINATION core/ebin)
