cmake_minimum_required(VERSION 3.25)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
endif()
project(windows_native_sdk LANGUAGES C)
if(NOT WIN32 OR NOT (MSVC OR MINGW))
    message(FATAL_ERROR "This fixture qualifies native Windows consumers")
endif()
find_package(BEAM CONFIG REQUIRED)
set(_module "${CMAKE_CURRENT_BINARY_DIR}/pkg_acceptance.beam")
add_custom_command(OUTPUT "${_module}"
    COMMAND $<TARGET_FILE:BEAM::CLI> compile --deterministic
        -o "${CMAKE_CURRENT_BINARY_DIR}" -- "${CMAKE_CURRENT_SOURCE_DIR}/pkg_acceptance.erl"
    DEPENDS BEAM::CLI "${CMAKE_CURRENT_SOURCE_DIR}/pkg_acceptance.erl" VERBATIM)
add_custom_target(module_bytes ALL DEPENDS "${_module}")
add_library(native_answer MODULE native.c)
beam_target_nif(native_answer MODULE pkg_acceptance)
add_library(builtin_answer STATIC native.c)
target_compile_definitions(builtin_answer PRIVATE NATIVE_ANSWER=23)
beam_target_nif(builtin_answer MODULE pkg_acceptance)
get_target_property(_registration builtin_answer BEAM_NIF_REGISTRATION)
get_target_property(_header builtin_answer BEAM_NIF_HEADER)
foreach(_mode shared static builtin)
    add_executable(consumer_${_mode} main.c)
    if(MINGW)
        target_link_options(consumer_${_mode} PRIVATE -municode)
    endif()
    add_dependencies(consumer_${_mode} module_bytes native_answer)
    target_compile_features(consumer_${_mode} PRIVATE c_std_11)
    beam_target_bytecode_verifier(consumer_${_mode} INVENTORIES services)
    if(_mode STREQUAL shared)
        target_link_libraries(consumer_${_mode} PRIVATE BEAM::Host BEAM::Runtime)
    else()
        target_link_libraries(consumer_${_mode} PRIVATE BEAM::Host BEAM::RuntimeStatic)
        target_compile_definitions(consumer_${_mode} PRIVATE STATIC_HOST)
    endif()
    if(_mode STREQUAL builtin)
        target_link_libraries(consumer_${_mode} PRIVATE builtin_answer)
        target_compile_definitions(consumer_${_mode} PRIVATE
            BUILTIN_HEADER="${_header}" BUILTIN_REGISTRATION=${_registration} EXPECTED_ANSWER=23)
    endif()
endforeach()
add_executable(deployment_probe deployment.c)
target_link_libraries(deployment_probe PRIVATE BEAM::Host BEAM::Runtime)
beam_target_deployment(deployment_probe PROFILE bare)
beam_install_deployment(deployment_probe)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/inputs.txt"
    CONTENT "${_module}\n$<TARGET_FILE:native_answer>\n$<TARGET_FILE_NAME:BEAM::Runtime>\n")
