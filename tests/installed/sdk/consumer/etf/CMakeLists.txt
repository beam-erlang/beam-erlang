cmake_minimum_required(VERSION 3.25)
project(etf_consumer LANGUAGES C CXX)
if(CODEC_ONLY)
    find_package(BEAM CONFIG REQUIRED COMPONENTS ETF)
    if(TARGET BEAM::Runtime OR TARGET BEAM::RuntimeStatic)
        message(FATAL_ERROR "Codec-only discovery imported a VM")
    endif()
else()
    find_package(BEAM CONFIG REQUIRED COMPONENTS Runtime RuntimeStatic CLI ETF)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/etf_echo.beam"
        COMMAND $<TARGET_FILE:BEAM::CLI> compile -o "${CMAKE_CURRENT_BINARY_DIR}" -- "${CMAKE_CURRENT_SOURCE_DIR}/etf_echo.erl"
        DEPENDS etf_echo.erl BEAM::CLI VERBATIM)
    add_custom_target(etf_module ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/etf_echo.beam")
endif()
add_library(codec STATIC codec.c)
beam_target_etf_objects(codec)
beam_target_etf_objects(codec) # Incorporation is idempotent.
foreach(language c cxx)
    if(CODEC_ONLY)
        add_executable(etf_${language} main.${language})
        target_compile_definitions(etf_${language} PRIVATE CODEC_ONLY)
        target_link_libraries(etf_${language} PRIVATE codec)
    else()
        foreach(form Runtime RuntimeStatic)
            add_executable(etf_${language}_${form} main.${language})
            target_link_libraries(etf_${language}_${form} PRIVATE codec BEAM::${form})
        endforeach()
    endif()
endforeach()
