cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug>:Embedded>")
project(beam_sdk_consumer LANGUAGES C CXX)
find_package(BEAM ${EXPECTED_VERSION} EXACT CONFIG REQUIRED)
set(_required_targets Runtime NIF Driver)
if(TARGET BEAM::RuntimeStatic)
    message(FATAL_ERROR "SDK must not publish the removed BEAM::RuntimeStatic target")
endif()
if(NOT WIN32)
    list(APPEND _required_targets ChildSetup)
elseif(TARGET BEAM::ChildSetup)
    message(FATAL_ERROR "Windows SDK must not publish a child helper")
endif()
foreach(_required IN LISTS _required_targets)
    if(NOT TARGET BEAM::${_required})
        message(FATAL_ERROR "Missing public SDK target BEAM::${_required}")
    endif()
endforeach()
foreach(_field VERSION ERTS_VERSION FLAVOR BUILD_TYPE BUILD_ID HOST_TRIPLE ARCHITECTURE SYSTEM_NAME UPSTREAM_COMMIT)
    if(NOT BEAM_SDK_${_field} STREQUAL EXPECTED_${_field})
        message(FATAL_ERROR "Unexpected SDK ${_field}: '${BEAM_SDK_${_field}}', expected '${EXPECTED_${_field}}'")
    endif()
endforeach()
if(NOT BEAM_SDK_DYNAMIC_TRACE STREQUAL "none")
    message(FATAL_ERROR "Unexpected SDK tracing personality")
endif()
foreach(_language c cpp)
    add_executable(beam_sdk_${_language} main.${_language})
    target_link_libraries(beam_sdk_${_language} PRIVATE BEAM::Runtime)
    install(TARGETS beam_sdk_${_language} RUNTIME DESTINATION bin)
endforeach()
if(WIN32)
    # The VM and canonical resources live in a DLL, while a separate executable
    # supplies the CRT entry. This catches executable-module resource lookup.
    add_library(beam_sdk_runtime MODULE dll_runtime.c)
    target_link_libraries(beam_sdk_runtime PRIVATE BEAM::Runtime)
    set_target_properties(beam_sdk_runtime PROPERTIES PREFIX "")
    add_executable(beam_sdk_dll dll_host.c)
    install(TARGETS beam_sdk_runtime beam_sdk_dll RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
endif()
add_library(beam_sdk_nif MODULE nif.c)
target_link_libraries(beam_sdk_nif PRIVATE BEAM::NIF ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(beam_sdk_nif PRIVATE _GNU_SOURCE)
endif()
add_library(beam_sdk_driver MODULE driver.c)
target_link_libraries(beam_sdk_driver PRIVATE BEAM::Driver)
set_target_properties(beam_sdk_nif beam_sdk_driver PROPERTIES PREFIX "")
if(NOT WIN32)
    set_target_properties(beam_sdk_nif beam_sdk_driver PROPERTIES SUFFIX ".so")
endif()
install(TARGETS beam_sdk_nif beam_sdk_driver LIBRARY DESTINATION native RUNTIME DESTINATION native)
if(NOT WIN32)
    beam_install_child_setup(DESTINATION bin)
endif()
beam_install_preloaded_modules(DESTINATION "lib/erts-${BEAM_SDK_ERTS_VERSION}/ebin")
beam_install_preloaded_sources(DESTINATION "lib/erts-${BEAM_SDK_ERTS_VERSION}/src")
beam_install_native_headers(DESTINATION include)
