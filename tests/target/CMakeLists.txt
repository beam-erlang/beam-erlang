# Acquired-SDK consumer also used by Android and iOS execution adapters.
cmake_minimum_required(VERSION 3.25)
project(beam_target_acceptance LANGUAGES C)
find_package(BEAM CONFIG REQUIRED COMPONENTS Runtime)
if(NOT EXISTS "${BEAM_TEST_ANSWER_HEADER}")
    message(FATAL_ERROR "Select the producer's source-built answer_beam.h")
endif()
get_filename_component(_generated "${BEAM_TEST_ANSWER_HEADER}" DIRECTORY)
add_executable(first-call ../installed/sdk/consumer/first-call/main.c)
target_include_directories(first-call PRIVATE "${_generated}")
target_link_libraries(first-call PRIVATE BEAM::Runtime)
set_target_properties(first-call PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib;@executable_path/Frameworks")
add_executable(api-validation ../checks/api_validation.c)
target_link_libraries(api-validation PRIVATE BEAM::Runtime)
set_target_properties(api-validation PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib;@executable_path/Frameworks")
add_executable(deployment_host ../installed/sdk/deployment/main.c)
target_link_libraries(deployment_host PRIVATE BEAM::Runtime)
target_include_directories(deployment_host PRIVATE "${_generated}" ../support)
target_compile_definitions(deployment_host PRIVATE DEPLOY_RETURNING BEAM_TARGET_ACCEPTANCE)
beam_target_bytecode_verifier(deployment_host INVENTORIES base boot services)
beam_target_deployment(deployment_host PROFILE beam-returning COMPONENTS file network zlib)
beam_install_deployment(deployment_host)
foreach(_id IN ITEMS runtime file network zlib)
    set(_artifact "component.${_id}")
    if(_id STREQUAL "runtime")
        set(_artifact runtime)
    endif()
    beam_get_runtime_artifact(_relative NAME "${_artifact}" RELATIVE)
    if(APPLE)
        string(REGEX REPLACE "^lib/" "Frameworks/" _relative "${_relative}")
    endif()
    string(TOUPPER "${_id}" _macro)
    target_compile_definitions(deployment_host PRIVATE "TARGET_${_macro}=\"${_relative}\"")
endforeach()
set_target_properties(deployment_host PROPERTIES BUILD_RPATH "$ORIGIN/../lib;@executable_path/Frameworks")
if(APPLE)
    beam_get_runtime_artifact(_runtime_binary NAME runtime)
    get_filename_component(_runtime_framework "${_runtime_binary}" DIRECTORY)
    foreach(_target IN ITEMS first-call api-validation deployment_host)
        set_target_properties(${_target} PROPERTIES MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_GUI_IDENTIFIER "org.erlangsters.acceptance.${_target}"
            XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
        add_custom_command(TARGET ${_target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:${_target}>/Frameworks"
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${_runtime_framework}"
                "$<TARGET_BUNDLE_DIR:${_target}>/Frameworks/BEAM.framework")
    endforeach()
endif()
