cmake_minimum_required(VERSION 3.25)
project(BEAMiOSAcceptance LANGUAGES C OBJC)
find_package(BEAM CONFIG REQUIRED COMPONENTS Runtime)
include("${BEAM_PACKAGE_EXPORT}/BEAMPackageExport.cmake")
add_executable(BEAMAcceptance main.m)
target_link_libraries(BEAMAcceptance PRIVATE BEAMPackageInputs "-framework UIKit" "-framework Foundation")
set_target_properties(BEAMAcceptance PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER org.erlangsters.acceptance.ios
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2")
if(BEAM_SIGNING_TEAM)
    set_target_properties(BEAMAcceptance PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${BEAM_SIGNING_TEAM}")
endif()
beam_target_ios_application(BEAMAcceptance COMPONENTS file network zlib INVENTORIES base services)
set(RESOURCE_VARIANT normal CACHE STRING "normal, missing or corrupt; applied before Xcode signs the app")
if(NOT RESOURCE_VARIANT MATCHES "^(normal|missing|corrupt)$")
    message(FATAL_ERROR "Unsupported resource variant")
endif()
foreach(_relative IN LISTS BEAM_PACKAGE_ASSETS)
    set(_input "${BEAM_PACKAGE_EXPORT_ROOT}/${_relative}")
    if(_relative MATCHES "/ebin/ios_acceptance.beam$")
        if(RESOURCE_VARIANT STREQUAL "missing")
            continue()
        elseif(RESOURCE_VARIANT STREQUAL "corrupt")
            set(_input "${CMAKE_CURRENT_BINARY_DIR}/corrupt/ios_acceptance.beam")
            file(WRITE "${_input}" "corrupt before signing")
        endif()
    endif()
    get_filename_component(_parent "${_relative}" DIRECTORY)
    target_sources(BEAMAcceptance PRIVATE "${_input}")
    set_source_files_properties("${_input}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${_parent}")
    if(_relative MATCHES "/priv/payload.txt$")
        target_compile_definitions(BEAMAcceptance PRIVATE "FIXTURE_RESOURCE=\"${_relative}\"")
    endif()
endforeach()
